  <action>
    Create `frontend/src/components/timeline/DraggableTimeBlock.tsx`.

    Add `'use client';` header.

    Imports:
    - `useRef` from 'react'
    - `useDragToReschedule, getHasTouched` from `@/hooks/useDragToReschedule`
    - `type PlannerItem` from `@/hooks/usePlannerTimeline`
    - `PX_PER_HOUR, BLOCK_MIN_HEIGHT` from `./timelineConstants`

    **Props interface (export it):**
    ```ts
    export interface DraggableTimeBlockProps {
      item: PlannerItem;
      hourHeight: number;
      compact?: boolean;
      onReschedule: (taskId: string, newStartAt: string) => Promise<void>;
      onResize: (taskId: string, newDurationMinutes: number) => Promise<void>;
      onDragActiveChange?: (active: boolean) => void;
    }
    ```

    **isNative guard:** inside the component, derive:
    ```ts
    const isNative = item.id.startsWith('native:');
    ```
    All pointer handlers check `if (!isNative) return;` before calling hook methods.

    **Block ref:** `const blockRef = useRef<HTMLDivElement>(null);`

    **Hook call:**
    ```ts
    const { dragState, liveTop, liveHeight, revertTransition, startMove, startResize } =
      useDragToReschedule({ item, onReschedule, onResize, onDragActiveChange });
    ```

    **State-derived class sets** (resolve to Tailwind strings):
    - `isDragMove = dragState === 'drag-move'`
    - `isDragResize = dragState === 'drag-resize'`
    - `isReverting = dragState === 'reverting'`
    - `isErrorTint = dragState === 'error-tint'`
    - `isTouchDevice = getHasTouched()`

    **Outer container classes** (static + conditional):
    ```
    `absolute border-l-[3px] rounded-r px-2 py-1 overflow-hidden left-14 right-2 bg-blue-50 group
     select-none cursor-grab
     ${isErrorTint ? 'border-red-400' : 'border-blue-500'}
     ${isDragMove ? 'shadow-md opacity-90' : ''}
     ${isDragResize ? 'shadow-sm' : ''}
     ${item.completed ? 'opacity-40' : ''}`
    ```

    **Outer container inline style** (runtime px values only — per project rules):
    ```ts
    const containerStyle: React.CSSProperties = {
      top: liveTop,
      height: liveHeight,
      zIndex: isDragMove ? 50 : isDragResize ? 20 : 10,
      touchAction: (isDragMove || isDragResize) ? 'none' : undefined,
      transition: isReverting ? revertTransition : 'none',
      cursor: isDragMove ? 'grabbing' : undefined,
    };
    ```

    **Pointer handlers on block body (not resize zone):**
    ```ts
    const handleMovePointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
      if (!isNative) return;
      if (!blockRef.current) return;
      startMove(e, blockRef.current);
    };
    ```

    **Pointer handler on resize zone:**
    ```ts
    const handleResizePointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
      e.stopPropagation(); // prevent block's handleMovePointerDown from also firing
      if (!isNative) return;
      if (!blockRef.current) return;
      startResize(e, blockRef.current);
    };
    ```

    **Block body content** (inherit text rendering from PlannerTimeBlock visual):
    - Title: `<p className="text-[0.7rem] font-semibold leading-tight text-blue-900 truncate">{item.title}</p>`
    - Time range (only when liveHeight >= 36): derive start/end strings from item.startAt + liveHeight-computed duration for display. Use same formatting as PlannerTimeBlock — inspect PlannerTimeBlock.tsx and replicate the time-range rendering exactly (do not invent new formatting).
    - Completed overlay: `{item.completed && <div className="absolute inset-0 bg-white/40" />}`

    **Resize handle** (absolutely positioned at block bottom):
    ```tsx
    <div
      className="absolute bottom-0 left-0 right-0 h-5 flex items-end justify-center pb-1 cursor-ns-resize select-none"
      onPointerDown={handleResizePointerDown}
      role="slider"
      aria-label="Drag to resize duration"
      aria-valuemin={15}
      aria-valuemax={240}
      aria-valuenow={item.durationMinutes ?? 30}
      aria-valuetext={`${item.durationMinutes ?? 30} minutes`}
    >
      <div className={`w-8 h-1 rounded-full bg-blue-300 transition-opacity duration-150 ${
        isDragResize
          ? 'opacity-100'
          : isTouchDevice
            ? 'opacity-60'
            : 'opacity-0 group-hover:opacity-100'
      }`} />
    </div>
    ```

    **Full JSX return:**
    ```tsx
    return (
      <div
        ref={blockRef}
        className={containerClasses}
        style={containerStyle}
        onPointerDown={handleMovePointerDown}
      >
        {/* title */}
        {/* time range — conditional on height */}
        {/* completed overlay */}
        {/* resize handle — ALWAYS last child */}
        {isNative && <div className="absolute bottom-0 left-0 right-0 h-5 ...">...</div>}
      </div>
    );
    ```
    Sync items (non-native) render without the resize handle (drag is no-op for them).

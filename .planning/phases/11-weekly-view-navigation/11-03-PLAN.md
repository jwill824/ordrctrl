    **Helper: getWeekRange string for N weeks from now**

    The test needs to compute the expected label text to assert against. Replicate the same logic
    used by `formatWeekRange` in the test file (do not import from src — e2e tests must be

    ```typescript
    function formatWeekRange(weekStart: Date): string {
      const end = new Date(weekStart);
      end.setDate(weekStart.getDate() + 6);
      const startLabel = weekStart.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
      const endLabel = end.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
      if (weekStart.getMonth() === end.getMonth()) {
        // same month: "Jun 1–7" (no spaces)
        const endDay = end.getDate();
        return `${startLabel}–${endDay}`;
      }
      // cross-month: "Jun 30 – Jul 6" (spaces)
      return `${startLabel} – ${endLabel}`;
    }

    function getWeekStart(date: Date): Date {
      const d = new Date(date);
      d.setHours(0, 0, 0, 0);
      const day = d.getDay();
      const diff = day === 0 ? -6 : 1 - day;
      d.setDate(d.getDate() + diff);
      return d;
    }

    function addDays(date: Date, n: number): Date {
      const d = new Date(date);
      d.setDate(d.getDate() + n);
      return d;
    }
    ```
